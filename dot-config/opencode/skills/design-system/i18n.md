# Internationalization

## Logical Properties

Use logical properties for RTL support:

| Physical | Logical |
|----------|---------|
| `margin-left` | `margin-inline-start` |
| `margin-right` | `margin-inline-end` |
| `padding-left` | `padding-inline-start` |
| `padding-right` | `padding-inline-end` |
| `border-left` | `border-inline-start` |
| `border-right` | `border-inline-end` |
| `text-align: left` | `text-align: start` |
| `text-align: right` | `text-align: end` |

## RTL Support

```css
[dir="rtl"] {
  direction: rtl;
}

/* Logical properties handle most cases automatically */
.button {
  padding-inline-start: var(--space-md);
  padding-inline-end: var(--space-md);
}

/* For cases requiring manual RTL */
[dir="rtl"] .icon-left {
  margin-right: var(--space-sm);
  margin-left: 0;
}

[dir="rtl"] .icon-right {
  margin-left: var(--space-sm);
  margin-right: 0;
}
```

## Text Expansion

Languages can expand 200%+ when translated:
- Use `min-width` instead of `fixed width`
- Use flexbox/grid for automatic resizing
- Test with pseudo-localization

```css
/* Good - flexible width */
.button {
  min-width: 100px;
  padding: var(--space-sm) var(--space-md);
}

/* Bad - fixed width */
.button {
  width: 150px;
  padding: var(--space-sm) var(--space-md);
}

/* Good - flexible container */
.card {
  min-width: 300px;
  max-width: 100%;
}

/* Good - flex layout */
.nav-item {
  display: flex;
  align-items: center;
  gap: var(--space-sm);
}
```

## Pluralization Rules

```javascript
// ICU Message Format for plurals
const messages = {
  en: {
    items: '{count, plural, =0 {No items} one {# item} other {# items}}',
    users: '{count, plural, =0 {No users} one {# user} other {# users}}'
  },
  ar: {
    items: '{count, plural, zero {لا عناصر} one {عنصر واحد} two {عنصران} few {# عناصر} many {# عنصرًا} other {# عنصر}}',
    users: '{count, plural, zero {لا مستخدمين} one {مستخدم واحد} two {مستخدمان} few {# مستخدمين} many {# مستخدمًا} other {# مستخدم}}'
  }
};

// Usage
const formatMessage = (key, count, locale) => {
  const message = messages[locale][key];
  // Use a library like messageformat to handle plurals
  return messageformat(message, { locale, count });
};
```

## RTL Service (Angular)

```typescript
import { Injectable, signal, effect, PLATFORM_ID, Inject } from '@angular/core';
import { isPlatformBrowser } from '@angular/common';

export type Direction = 'ltr' | 'rtl';

@Injectable({ providedIn: 'root' })
export class RTLService {
  private readonly STORAGE_KEY = 'direction';

  readonly direction = signal<Direction>(this.getInitialDirection());

  constructor(@Inject(PLATFORM_ID) private platformId: Object) {
    effect(() => {
      const dir = this.direction();
      if (isPlatformBrowser(this.platformId)) {
        document.documentElement.setAttribute('dir', dir);
        localStorage.setItem(this.STORAGE_KEY, dir);
      }
    });

    if (isPlatformBrowser(this.platformId)) {
      document.documentElement.setAttribute('dir', this.direction());
    }
  }

  private getInitialDirection(): Direction {
    if (isPlatformBrowser(this.platformId)) {
      const stored = localStorage.getItem(this.STORAGE_KEY) as Direction | null;
      if (stored === 'ltr' || stored === 'rtl') return stored;
    }
    return 'ltr';
  }

  isRTL(): boolean {
    return this.direction() === 'rtl';
  }

  setDirection(direction: Direction): void {
    this.direction.set(direction);
  }

  toggleDirection(): void {
    this.direction.update(d => d === 'ltr' ? 'rtl' : 'ltr');
  }
}
```

### Usage in Component

```typescript
import { Component, inject, ChangeDetectionStrategy } from '@angular/core';
import { RTLService } from './rtl.service';

@Component({
  selector: 'app-direction-toggle',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <button (click)="rtl.toggleDirection()">
      {{ rtl.isRTL() ? 'Switch to LTR' : 'Switch to RTL' }}
    </button>
  `
})
export class DirectionToggleComponent {
  rtl = inject(RTLService);
}
```

## Accessibility for Internationalization

- Use `lang` attribute on HTML element and content sections
- Use `dir` attribute for text direction
- Ensure focus order is correct in RTL layouts
- Test with screen readers in different languages
- Provide language switcher in accessible location
- Announce language changes to screen readers

```html
<html lang="en" dir="ltr">
<head>
  <meta charset="UTF-8">
  <title>Internationalized Application</title>
</head>
<body>
  <header>
    <nav aria-label="Main navigation">
      <!-- Navigation items -->
    </nav>
    <div class="language-switcher" role="group" aria-label="Language selection">
      <button aria-label="Switch to Arabic" data-lang="ar">عربي</button>
      <button aria-label="Switch to English" data-lang="en">English</button>
    </div>
  </header>
  
  <main>
    <section lang="en" dir="ltr">
      <h1>English Content</h1>
      <p>This content is in English.</p>
    </section>
    
    <section lang="ar" dir="rtl">
      <h1>محتوى عربي</h1>
      <p>هذا المحتوى بالعربية.</p>
    </section>
  </main>
</body>
</html>
```
